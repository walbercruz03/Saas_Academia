const db = require('../config/database');
const bcrypt = require('bcryptjs');

class Usuario {
  static criarTabela(callback) {
    const sql = `
      CREATE TABLE IF NOT EXISTS usuarios (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nome TEXT NOT NULL,
        idade INTEGER,
        cpf TEXT,
        email_ou_matricula TEXT UNIQUE NOT NULL,
        telefone TEXT,
        senha_hash TEXT NOT NULL,
        cargo TEXT NOT NULL CHECK(cargo IN ('admin', 'professor')),
        status TEXT NOT NULL DEFAULT 'ativo' CHECK(status IN ('ativo', 'inativo')),
        criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `;
    db.run(sql, callback);
  }

  static create({ nome, idade, cpf, email_ou_matricula, telefone, senha, cargo }, callback) {
    const senhaHash = bcrypt.hashSync(senha, 10);
    const sql = `INSERT INTO usuarios (nome, idade, cpf, email_ou_matricula, telefone, senha_hash, cargo) VALUES (?, ?, ?, ?, ?, ?, ?)`;
    db.run(sql, [nome, idade, cpf, email_ou_matricula, telefone, senhaHash, cargo], function (err) {
      callback(err, this);
    });
  }

  static findByEmailOuMatricula(email_ou_matricula, callback) {
    const sql = `SELECT * FROM usuarios WHERE email_ou_matricula = ?`;
    db.get(sql, [email_ou_matricula], callback);
  }

  static findById(id, callback) {
    const sql = `SELECT id, nome, idade, cpf, email_ou_matricula, telefone, cargo, status, criado_em FROM usuarios WHERE id = ?`;
    db.get(sql, [id], callback);
  }

  static listar(callback) {
    const sql = `SELECT id, nome, idade, cpf, email_ou_matricula, telefone, cargo, status, criado_em FROM usuarios ORDER BY criado_em DESC`;
    db.all(sql, [], callback);
  }

  static atualizarStatus(id, status, callback) {
    const sql = `UPDATE usuarios SET status = ? WHERE id = ?`;
    db.run(sql, [status, id], callback);
  }

  static validarSenha(senha, senhaHash) {
    return bcrypt.compareSync(senha, senhaHash);
  }
}

module.exports = Usuario;

      return callback(null, []);

    const ids = modelos.map((m) => m.id);
    const placeholders = ids.map(() => '?').join(', ');

    // 1. Busca todas as divisões dos modelos

      SELECT
        dt.id,
        dt.modelo_id,
        dt.nome,
        dt.descricao,
        dt.ordem
      FROM divisoes_treino dt
      WHERE dt.modelo_id IN (${placeholders})
      ORDER BY dt.ordem ASC, dt.id ASC

    db.all(sqlDivisoes, ids, (divErr, rowsDivisoes) => {
      if (divErr) return callback(divErr);

      // 2. Busca todos os exercícios vinculados aos modelos
      const sqlExercicios = `
        SELECT
          mte.id,
          mte.modelo_id,
          mte.divisao_id,
          mte.exercicio_id,
          mte.ordem,
          mte.series,
          mte.repeticoes,
          mte.carga,
          mte.descanso,
          mte.observacao,
          e.nome AS exercicio_nome,
          e.equipamento AS exercicio_equipamento,
          e.dificuldade AS exercicio_dificuldade,
          e.imagem_url AS exercicio_imagem_url,
          e.video_url AS exercicio_video_url
        FROM modelo_treino_exercicios mte
        INNER JOIN exercicios e ON e.id = mte.exercicio_id
        WHERE mte.modelo_id IN (${placeholders})
        ORDER BY mte.ordem ASC, mte.id ASC
      `;

      db.all(sqlExercicios, ids, (exErr, rowsExercicios) => {
        if (exErr) return callback(exErr);

        const mapaModelos = new Map();

            ...m,
            divisoes: [],
            exercicios: [],
          });
        });

        // Mapeia divisões por modelo e prepara mapa de divisões
        const mapaDivisoes = new Map();
        (rowsDivisoes || []).forEach((d) => {
          const mod = mapaModelos.get(d.modelo_id);
          const divisaoObj = {
            id: d.id,
            modelo_id: d.modelo_id,
            nome: d.nome,
            descricao: d.descricao || '',
            ordem: Number(d.ordem || 1),
            exercicios: [],
          };
          mapaDivisoes.set(d.id, divisaoObj);
          if (mod) {
            mod.divisoes.push(divisaoObj);
          }
        });

        // Se algum modelo não tiver divisões cadastradas, cria uma virtual padrão
        mapaModelos.forEach((mod) => {
          if (mod.divisoes.length === 0) {
            const divPadrao = {
              id: null,
              modelo_id: mod.id,
              nome: mod.divisao || 'Treino A',
              descricao: 'Divisão Geral',
              ordem: 1,
              exercicios: [],
            };
            mod.divisoes.push(divPadrao);
          }
        });

        // Associa exercícios às suas divisões correspondentes
        (rowsExercicios || []).forEach((item) => {
          const mod = mapaModelos.get(item.modelo_id);
          if (!mod) return;

          const exercicioItem = {
            id: item.id,
            modelo_id: item.modelo_id,
            divisao_id: item.divisao_id,
            exercicio_id: item.exercicio_id,
            nome: item.exercicio_nome,
            equipamento: item.exercicio_equipamento,
            dificuldade: item.exercicio_dificuldade,
            imagem_url: item.exercicio_imagem_url,
            video_url: item.exercicio_video_url,
            ordem: Number(item.ordem || 1),
            series: item.series || '3',
            repeticoes: item.repeticoes || '12',
            carga: item.carga || '',
            descanso: item.descanso || '60s',
            observacao: item.observacao || '',
          };

          // Adiciona na lista consolidada do modelo para compatibilidade
          mod.exercicios.push(exercicioItem);

          // Associa à divisão específica
          let targetDiv = item.divisao_id ? mapaDivisoes.get(item.divisao_id) : null;
          if (!targetDiv) {
            targetDiv = mod.divisoes[0];
          }
          if (targetDiv) {
            targetDiv.exercicios.push(exercicioItem);
          }
        });

        callback(null, Array.from(mapaModelos.values()));
      });
    });

const db = require('../config/database');

const Exercicio = {
  criarTabela: (callback) => {
    const query = `
      CREATE TABLE IF NOT EXISTS exercicios (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nome TEXT NOT NULL,
        descricao TEXT,
        equipamento TEXT,
        dificuldade TEXT NOT NULL DEFAULT 'intermediario',
        imagem_url TEXT,
        video_url TEXT,
        ativo INTEGER NOT NULL DEFAULT 1,
        criado_em DATETIME DEFAULT CURRENT_TIMESTAMP
      )
    `;
    db.run(query, (err) => {
      if (err) return callback(err);

      const joinQuery = `
        CREATE TABLE IF NOT EXISTS exercicio_grupos_musculares (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          exercicio_id INTEGER NOT NULL,
          grupo_muscular_id INTEGER NOT NULL,
          papel TEXT NOT NULL DEFAULT 'principal',
          FOREIGN KEY (exercicio_id) REFERENCES exercicios (id) ON DELETE CASCADE,
          FOREIGN KEY (grupo_muscular_id) REFERENCES grupos_musculares (id) ON DELETE CASCADE,
          UNIQUE (exercicio_id, grupo_muscular_id, papel)
        )
      `;
      db.run(joinQuery, callback);
    });
  },

  sincronizarGrupos: (exercicioId, gruposPrincipais = [], gruposSecundarios = [], callback) => {
    const principaisIds = [...new Set((gruposPrincipais || []).map(Number).filter(Boolean))];
    const secundariosIds = [...new Set((gruposSecundarios || []).map(Number).filter(Boolean))];

    db.run(`DELETE FROM exercicio_grupos_musculares WHERE exercicio_id = ?`, [exercicioId], (err) => {
      if (err) return callback(err);

      const itens = [
        ...principaisIds.map((grupoId) => ({ grupoId, papel: 'principal' })),
        ...secundariosIds.map((grupoId) => ({ grupoId, papel: 'secundario' })),
      ];

      if (itens.length === 0) return callback(null);

      let concluídos = 0;
      let falhou = false;

      itens.forEach((item) => {
        if (falhou) return;
        db.run(
          `INSERT INTO exercicio_grupos_musculares (exercicio_id, grupo_muscular_id, papel) VALUES (?, ?, ?)`,
          [exercicioId, item.grupoId, item.papel],
          (insertErr) => {
            if (falhou) return;
            if (insertErr) {
              falhou = true;
              return callback(insertErr);
            }
            concluídos += 1;
            if (concluídos === itens.length) {
              callback(null);
            }
          }
        );
      });
    });
  },

  anexarGrupos: (exercicios, callback) => {
    if (!Array.isArray(exercicios) || exercicios.length === 0) {
      return callback(null, []);
    }

    const ids = exercicios.map((e) => e.id);
    const placeholders = ids.map(() => '?').join(', ');

    const sql = `
      SELECT
        egm.exercicio_id,
        egm.papel,
        gm.id AS grupo_id,
        gm.nome AS grupo_nome
      FROM exercicio_grupos_musculares egm
      INNER JOIN grupos_musculares gm ON gm.id = egm.grupo_muscular_id
      WHERE egm.exercicio_id IN (${placeholders})
      ORDER BY gm.nome ASC
    `;

    db.all(sql, ids, (err, vinculos) => {
      if (err) return callback(err);

      const mapa = new Map();
      exercicios.forEach((ex) => {
        mapa.set(ex.id, {
          ...ex,
          grupos_principais: [],
          grupos_secundarios: [],
        });
      });

      (vinculos || []).forEach((v) => {
        const alvo = mapa.get(v.exercicio_id);
        if (!alvo) return;
        const grupoObj = { id: v.grupo_id, nome: v.grupo_nome, papel: v.papel };
        if (v.papel === 'secundario') {
          alvo.grupos_secundarios.push(grupoObj);
        } else {
          alvo.grupos_principais.push(grupoObj);
        }
      });

      callback(null, Array.from(mapa.values()));
    });
  },

  listar: ({ busca = '', equipamento = '', dificuldade = '', grupoMuscularId = '', apenasAtivos = false } = {}, callback) => {
    const params = [];
    const conditions = [];
    let sql = `SELECT DISTINCT e.* FROM exercicios e`;

    if (grupoMuscularId && String(grupoMuscularId).trim() !== '') {
      sql += ` INNER JOIN exercicio_grupos_musculares egm_filtro ON egm_filtro.exercicio_id = e.id`;
      conditions.push(`egm_filtro.grupo_muscular_id = ?`);
      params.push(Number(grupoMuscularId));
    }

    if (busca && busca.trim() !== '') {
      const termo = `%${busca.trim()}%`;
      conditions.push(`(LOWER(e.nome) LIKE LOWER(?) OR LOWER(COALESCE(e.descricao, '')) LIKE LOWER(?) OR LOWER(COALESCE(e.equipamento, '')) LIKE LOWER(?))`);
      params.push(termo, termo, termo);
    }

    if (equipamento && equipamento.trim() !== '') {
      conditions.push(`LOWER(COALESCE(e.equipamento, '')) LIKE LOWER(?)`);
      params.push(`%${equipamento.trim()}%`);
    }

    if (dificuldade && dificuldade.trim() !== '') {
      conditions.push(`LOWER(e.dificuldade) = LOWER(?)`);
      params.push(dificuldade.trim());
    }

    if (apenasAtivos) {
      conditions.push(`e.ativo = 1`);
    }

    if (conditions.length > 0) {
      sql += ` WHERE ` + conditions.join(' AND ');
    }

    sql += ` ORDER BY e.nome ASC, e.id DESC`;

    db.all(sql, params, (err, rows) => {
      if (err) return callback(err);
      Exercicio.anexarGrupos(rows || [], callback);
    });
  },

  buscarPorId: (id, callback) => {
    db.get(`SELECT * FROM exercicios WHERE id = ?`, [id], (err, row) => {
      if (err) return callback(err);
      if (!row) return callback(null, null);
      Exercicio.anexarGrupos([row], (grupoErr, lista) => {
        if (grupoErr) return callback(grupoErr);
        callback(null, lista[0] || null);
      });
    });
  },

  criar: (dados, callback) => {
    const {
      nome,
      descricao = null,
      instrucoes = null,
      equipamento = null,
      dificuldade = 'intermediario',

      video_url = null,

      grupos_principais = [],
      grupos_secundarios = [],
    } = dados;

    const sql = `
      INSERT INTO exercicios (nome, descricao, instrucoes, equipamento, dificuldade, imagem_url, video_url, ativo)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `;

    const params = [
      nome.trim(),
      descricao ? descricao.trim() : null,
      instrucoes ? instrucoes.trim() : null,
      equipamento ? equipamento.trim() : null,
      dificuldade || 'intermediario',
      imagem_url ? imagem_url.trim() : null,
      video_url ? video_url.trim() : null,
      ativo === false || ativo === 0 ? 0 : 1,
    ];

    db.run(sql, params, function (err) {
      if (err) return callback(err);
      const novoId = this && this.lastID ? this.lastID : null;
      if (!novoId) return callback(new Error('Não foi possível obter o ID do exercício criado.'));

      Exercicio.sincronizarGrupos(novoId, grupos_principais, grupos_secundarios, (syncErr) => {
        if (syncErr) return callback(syncErr);
        Exercicio.buscarPorId(novoId, callback);
      });
    });
  },

  atualizar: (id, dados, callback) => {
    const {
      nome,
      descricao = null,
      instrucoes = null,
      equipamento = null,
      dificuldade = 'intermediario',
      imagem_url = null,
      video_url = null,
      ativo = 1,
      grupos_principais = [],
      grupos_secundarios = [],
    } = dados;

    const sql = `
      UPDATE exercicios
      SET nome = ?, descricao = ?, instrucoes = ?, equipamento = ?, dificuldade = ?, imagem_url = ?, video_url = ?, ativo = ?
      WHERE id = ?
    `;

    const params = [
      nome.trim(),
      descricao ? descricao.trim() : null,
      instrucoes ? instrucoes.trim() : null,
      equipamento ? equipamento.trim() : null,
      dificuldade || 'intermediario',
      imagem_url ? imagem_url.trim() : null,
      video_url ? video_url.trim() : null,
      ativo === false || ativo === 0 ? 0 : 1,
      id,
    ];

    db.run(sql, params, (err) => {
      if (err) return callback(err);
      Exercicio.sincronizarGrupos(id, grupos_principais, grupos_secundarios, (syncErr) => {
        if (syncErr) return callback(syncErr);
        Exercicio.buscarPorId(id, callback);
      });
    });
  },

  deletar: (id, callback) => {
    db.run(`DELETE FROM exercicios WHERE id = ?`, [id], function (err) {
      if (err) return callback(err);
      callback(null, {
        deleted: true,
        changes: this && typeof this.changes === 'number' ? this.changes : 0,
      });
    });
  },
};

const { Pool } = require("pg");
const path = require("path");

const isExplicitSqlite = process.env.DB_CLIENT === "sqlite" || process.env.USE_SQLITE === "true";
const hasDatabaseUrl = Boolean(process.env.DATABASE_URL);

let useSqlite = isExplicitSqlite || !hasDatabaseUrl;
let sqliteDb = null;

function getSqliteDb() {
  if (!sqliteDb) {
    const sqlite3 = require("sqlite3").verbose();
    const dbPath = path.resolve(__dirname, "..", "database.sqlite");
    sqliteDb = new sqlite3.Database(dbPath, (err) => {
      if (err) console.error("Erro ao conectar ao SQLite:", err.message);
      else console.log("Conectado ao banco de dados SQLite.");
    });
  }
  return sqliteDb;
}

let pool = null;
if (!useSqlite) {
  pool = new Pool({
    connectionString: process.env.DATABASE_URL,
    connectionTimeoutMillis: 3000,
  });

  pool.query("SELECT 1")
    .then(() => {
      console.log("Conectado ao banco de dados PostgreSQL.");
    })
    .catch((err) => {
      console.warn("PostgreSQL indisponível (" + err.message + "). Ativando fallback para SQLite local.");
      useSqlite = true;
      getSqliteDb();
    });
} else {
  getSqliteDb();
}

function traduzirSql(sql) {
  let s = String(sql)
    .replace(/INTEGER\s+PRIMARY\s+KEY\s+AUTOINCREMENT/gi, "SERIAL PRIMARY KEY")
    .replace(/\bDATETIME\b/gi, "TIMESTAMP")
    .replace(/\bBLOB\b/gi, "BYTEA")
    .replace(/datetime\(\s*'now'\s*(?:,\s*[^')]+'[^')]*')*\s*\)/gi, "CURRENT_TIMESTAMP")
    .replace(/date\(\s*'now'\s*(?:,\s*[^')]+'[^')]*')*\s*\)/gi, "CURRENT_DATE")
    .replace(/datetime\(\s*([^(),]+?)\s*\)/gi, "($1)::timestamp")
    .replace(/date\(\s*([^(),]+?)\s*\)/gi, "($1)::date");

  // Compatibilidade básica com consultas a sqlite_master
  if (/sqlite_master/i.test(s)) {
    s = s
      .replace(/\bsqlite_master\b/gi, "information_schema.tables")
      .replace(/\bname\b/gi, "table_name")
      .replace(/type\s*=\s*'table'\s+AND\s+/gi, "table_schema = 'public' AND table_type = 'BASE TABLE' AND ")
      .replace(/WHERE\s+type\s*=\s*'table'/gi, "WHERE table_schema = 'public' AND table_type = 'BASE TABLE'");
  }

  let idx = 0, sq = false, dq = false, out = "";
  for (let i = 0; i < s.length; i++) {
    const c = s[i];
    if (c === "'" && !dq) { sq = !sq; out += c; }
    else if (c === "\"" && !sq) { dq = !dq; out += c; }
    else if (c === "?" && !sq && !dq) { out += "$" + (++idx); }
    else { out += c; }
  }
  return out.trim();
}

function args(params, cb) {
  if (typeof params === "function") return { v: [], cb: params };
  return { v: Array.isArray(params) ? params : params !== undefined ? [params] : [], cb: typeof cb === "function" ? cb : null };
}

let fila = Promise.resolve();
const runQueue = (fn) => { const p = fila.then(fn, fn); fila = p.catch(() => {}); return p; };

const db = {
  pool,
  get isSqlite() { return useSqlite; },
  serialize(fn) {
    if (useSqlite) {
      getSqliteDb().serialize(fn);
    } else {
      if (typeof fn === "function") fn();
    }
  },
  run(sql, params, callback) {
    const { v, cb } = args(params, callback);
    if (useSqlite) {
      getSqliteDb().run(sql, v, function (err) {
        if (cb) cb.call(this, err);
      });
      return this;
    }

    let q = traduzirSql(sql);
    const isIns = /^\s*INSERT\s+INTO\b/i.test(q);
    const hasRet = /\bRETURNING\b/i.test(q);
    if (isIns && !hasRet) q = q.replace(/;?\s*$/, " RETURNING id");
    runQueue(async () => {
      try {
        const r = await pool.query(q, v);
        const ctx = { lastID: r.rows?.[0]?.id ?? null, changes: r.rowCount ?? 0 };
        if (cb) cb.call(ctx, null);
      } catch (err) {
        if (isIns && !hasRet && /column "id" does not exist/i.test(err.message)) {
          try {
            const r2 = await pool.query(traduzirSql(sql), v);
            const ctx2 = { lastID: null, changes: r2.rowCount ?? 0 };
            if (cb) cb.call(ctx2, null);
            return;
          } catch (e2) {
            if (cb) cb.call({ lastID: null, changes: 0 }, e2);
            else console.error("Erro SQL (run):", e2.message);
            return;
          }
        }
        if (cb) cb.call({ lastID: null, changes: 0 }, err);
        else console.error("Erro SQL (run):", err.message);
      }
    });
    return this;
  },
  get(sql, params, callback) {
    const { v, cb } = args(params, callback);
    if (useSqlite) {
      getSqliteDb().get(sql, v, cb);
      return this;
    }
    runQueue(async () => {
      try { const r = await pool.query(traduzirSql(sql), v); if (cb) cb(null, r.rows[0]); }
      catch (err) { if (cb) cb(err, undefined); else console.error("Erro SQL (get):", err.message); }
    });
    return this;
  },
  all(sql, params, callback) {
    const { v, cb } = args(params, callback);
    if (useSqlite) {
      getSqliteDb().all(sql, v, cb);
      return this;
    }
    runQueue(async () => {
      try { const r = await pool.query(traduzirSql(sql), v); if (cb) cb(null, r.rows || []); }
      catch (err) { if (cb) cb(err, []); else console.error("Erro SQL (all):", err.message); }
    });
    return this;
  },
};

module.exports = db;
